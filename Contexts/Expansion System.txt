### 4. Procedural Generation Manager

**Create a centralized system to orchestrate chunk generation:**

```csharp
public class ProceduralChunkGenerator : MonoBehaviour
{
    [Header("Generation Settings")]
    public List<BiomeGenerationRules> availableBiomes;
    public int globalSeed;
    public Vector2Int standardChunkSize = new Vector2Int(16, 16);
    
    [Header("Validation Settings")]
    public int maxGenerationAttempts = 5;
    public float minPathConnectivity = 0.8f;
    public int requiredBuildableSpaces = 10;
    
    public ProceduralExpansionChunk GenerateChunk(BiomeGenerationRules rules, 
        List<Vector2Int> connectionPoints, int expansionIndex)
    {
        int chunkSeed = globalSeed + expansionIndex;
        
        for (int attempt = 0; attempt < maxGenerationAttempts; attempt++)
        {
            var chunk = GenerateChunkAttempt(rules, connectionPoints, chunkSeed + attempt);
            
            if (ValidateChunk(chunk))
            {
                return chunk;
            }
        }
        
        // Fallback to simpler generation rules
        return GenerateFallbackChunk(rules, connectionPoints, chunkSeed);
    }
    
    private ProceduralExpansionChunk GenerateChunkAttempt(BiomeGenerationRules rules, 
        List<Vector2Int> connectionPoints, int seed)
    {
        // Phase 1: Base terrain
        var terrainData = terrainGenerator.GenerateBaseTerrain(rules, seed);
        
        // Phase 2: Path network
        var pathData = pathGenerator.GeneratePathNetwork(rules, connectionPoints, seed);
        
        // Phase 3: Obstacles
        obstacleGenerator.PlaceObstacles(terrainData, rules, pathData, seed);
        
        // Phase 4: Decoratives
        decorativeGenerator.AddDecorativeElements(terrainData, rules, seed);
        
        return CreateChunkFromData(terrainData, pathData, rules);
    }
    
    private bool ValidateChunk(ProceduralExpansionChunk chunk)
    {
        // Validate path connectivity
        if (!pathfindingValidator.ValidateConnectivity(chunk.generatedTileData, chunk.connectionPoints))
            return false;
            
        // Validate buildable spaces
        if (chunk.generatedBuildableAreas.Count < requiredBuildableSpaces)
            return false;
            
        // Validate strategic balance
        if (!strategyValidator.ValidateStrategicBalance(chunk))
            return false;
            
        return true;
    }
}
```

### 5. Integration with Existing Expansion System

**Modify your existing ExpansionManager to use procedural generation:**

```csharp
public class ExpansionManager : MonoBehaviour
{
    [Header("Procedural Generation")]
    public ProceduralChunkGenerator chunkGenerator;
    public List<BiomeGenerationRules> availableBiomes;
    
    private void OnWaveCompleted()
    {
        // Generate 2-3 random biome options
        var expansionOptions = GenerateExpansionOptions();
        
        // Show UI for player selection
        expansionUI.ShowOptions(expansionOptions);
        
        // Pause game
        Time.timeScale = 0f;
    }
    
    private List<ExpansionOption> GenerateExpansionOptions()
    {
        var options = new List<ExpansionOption>();
        var selectedBiomes = SelectRandomBiomes(3);
        
        foreach (var biome in selectedBiomes)
        {
            var connectionPoints = CalculateConnectionPoints();
            var proceduralChunk = chunkGenerator.GenerateChunk(biome, connectionPoints, currentExpansionCount);
            
            var option = new ExpansionOption
            {
                optionName = biome.biomeName,
                proceduralChunk = proceduralChunk,
                pathLengthModifier = CalculatePathLength(proceduralChunk),
                buildSpaceModifier = CalculateBuildSpaces(proceduralChunk),
                riskAssessment = CalculateRisk(proceduralChunk),
                previewImage = GeneratePreviewImage(proceduralChunk),
                strategicDescription = GenerateDescription(proceduralChunk)
            };
            
            options.Add(option);
        }
        
        return options;
    }
}
```
